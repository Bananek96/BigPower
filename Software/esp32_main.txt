#include <WiFi.h>
#include <WebServer.h>
#include <HardwareSerial.h>
#include <ArduinoJson.h>

// Konfiguracja UART do komunikacji z STM32
HardwareSerial SerialSTM32(1);
#define STM32_RX_PIN 16
#define STM32_TX_PIN 17

// Konfiguracja WiFi
const char* ssid = "Falownik_ESP32";
const char* password = "password123";

// Web server
WebServer server(80);

// Struktura ramki UART (taka sama jak w STM32)
#pragma pack(push, 1)
typedef struct {
    uint8_t start;     // 0xA5
    uint8_t command;   // 0x01-0xFF
    uint8_t length;    // 0-32
    uint8_t data[32];  // Dane polecenia
    uint8_t checksum;  // XOR wszystkich pól
    uint8_t stop;      // 0x5A
} UART_Frame;
#pragma pack(pop)

// Komendy (jak w STM32)
#define CMD_SET_FREQ  0x01
#define CMD_SET_MOD   0x02
#define CMD_START     0x10
#define CMD_STOP      0x11
#define CMD_GET_TELEMETRY 0x20

// Zmienne globalne
uint32_t currentFrequency = 5000;
uint8_t currentModulation = 0;
bool systemRunning = false;
float outputVoltage = 0.0f;
float outputCurrent = 0.0f;
float temperature = 25.0f;

void setup() {
    // Inicjalizacja UART do STM32
    SerialSTM32.begin(115200, SERIAL_8N1, STM32_RX_PIN, STM32_TX_PIN);
    
    // Inicjalizacja Serial dla debugowania
    Serial.begin(115200);
    
    // Konfiguracja WiFi
    WiFi.softAP(ssid, password);
    IPAddress IP = WiFi.softAPIP();
    Serial.print("AP IP address: ");
    Serial.println(IP);
    
    // Konfiguracja endpointów HTTP
    server.on("/", HTTP_GET, handleRoot);
    server.on("/api/start", HTTP_POST, handleStart);
    server.on("/api/stop", HTTP_POST, handleStop);
    server.on("/api/set_frequency", HTTP_POST, handleSetFrequency);
    server.on("/api/set_modulation", HTTP_POST, handleSetModulation);
    server.on("/api/status", HTTP_GET, handleStatus);
    
    server.begin();
    Serial.println("HTTP server started");
}

void loop() {
    server.handleClient();
    handleUARTResponses();
    delay(10);
}

// Obsługa odpowiedzi od STM32
void handleUARTResponses() {
    static UART_Frame rxFrame;
    static uint8_t rxIndex = 0;
    static uint8_t rxBuffer[sizeof(UART_Frame)];
    
    while (SerialSTM32.available()) {
        uint8_t byte = SerialSTM32.read();
        
        if (rxIndex == 0 && byte != 0xA5) continue; // Czekaj na start bajt
        
        rxBuffer[rxIndex++] = byte;
        
        if (rxIndex >= sizeof(UART_Frame)) {
            memcpy(&rxFrame, rxBuffer, sizeof(UART_Frame));
            
            // Sprawdź sumę kontrolną i stop bajt
            if (rxFrame.stop == 0x5A && calculateChecksum(&rxFrame) == rxFrame.checksum) {
                processSTM32Response(&rxFrame);
            }
            
            rxIndex = 0;
        }
    }
}

// Przetwarzanie odpowiedzi od STM32
void processSTM32Response(UART_Frame* frame) {
    switch(frame->command) {
        case CMD_GET_TELEMETRY:
            if (frame->length == 12) {
                memcpy(&currentFrequency, &frame->data[0], 4);
                memcpy(&outputVoltage, &frame->data[4], 4);
                memcpy(&outputCurrent, &frame->data[8], 4);
            }
            break;
            
        // Można dodać obsługę innych odpowiedzi
    }
}

// Wysyłanie komend do STM32
void sendUARTCommand(uint8_t command, uint8_t* data, uint8_t length) {
    UART_Frame frame;
    frame.start = 0xA5;
    frame.command = command;
    frame.length = length;
    memcpy(frame.data, data, length);
    frame.checksum = calculateChecksum(&frame);
    frame.stop = 0x5A;
    
    SerialSTM32.write((uint8_t*)&frame, sizeof(frame));
}

// Obliczanie sumy kontrolnej
uint8_t calculateChecksum(UART_Frame* frame) {
    uint8_t checksum = 0;
    checksum ^= frame->start;
    checksum ^= frame->command;
    checksum ^= frame->length;
    
    for (uint8_t i = 0; i < frame->length; i++) {
        checksum ^= frame->data[i];
    }
    
    return checksum;
}

// Handler dla strony głównej
void handleRoot() {
    String html = "<html><head><title>Sterowanie Falownikiem</title></head><body>";
    html += "<h1>Sterowanie Falownikiem Jednofazowym</h1>";
    html += "<p>System zdalnego sterowania falownikiem</p>";
    html += "<p>Status: " + String(systemRunning ? "Działa" : "Zatrzymany") + "</p>";
    html += "<p>Częstotliwość: " + String(currentFrequency) + " Hz</p>";
    html += "<p>Napięcie wyjściowe: " + String(outputVoltage) + " V</p>";
    html += "<p>Prąd wyjściowy: " + String(outputCurrent) + " A</p>";
    html += "<p>Temperatura: " + String(temperature) + " °C</p>";
    html += "</body></html>";
    
    server.send(200, "text/html", html);
}

// Handler dla startu falownika
void handleStart() {
    uint8_t data = 0;
    sendUARTCommand(CMD_START, &data, 1);
    systemRunning = true;
    
    server.send(200, "application/json", "{\"status\":\"success\",\"message\":\"Falownik uruchomiony\"}");
}

// Handler dla stopu falownika
void handleStop() {
    uint8_t data = 0;
    sendUARTCommand(CMD_STOP, &data, 1);
    systemRunning = false;
    
    server.send(200, "application/json", "{\"status\":\"success\",\"message\":\"Falownik zatrzymany\"}");
}

// Handler dla ustawienia częstotliwości
void handleSetFrequency() {
    if (server.hasArg("plain")) {
        String freqStr = server.arg("plain");
        uint16_t freq = freqStr.toInt();
        
        if (freq >= 1000 && freq <= 20000) {
            uint8_t data[2] = {(uint8_t)(freq >> 8), (uint8_t)(freq & 0xFF)};
            sendUARTCommand(CMD_SET_FREQ, data, 2);
            currentFrequency = freq;
            
            server.send(200, "application/json", "{\"status\":\"success\",\"message\":\"Częstotliwość ustawiona\"}");
            return;
        }
    }
    
    server.send(400, "application/json", "{\"status\":\"error\",\"message\":\"Nieprawidłowa częstotliwość\"}");
}

// Handler dla ustawienia modulacji
void handleSetModulation() {
    if (server.hasArg("plain")) {
        String modStr = server.arg("plain");
        uint8_t modType = modStr.toInt();
        
        if (modType == 0 || modType == 1) {
            uint8_t data[1] = {modType};
            sendUARTCommand(CMD_SET_MOD, data, 1);
            currentModulation = modType;
            
            server.send(200, "application/json", "{\"status\":\"success\",\"message\":\"Typ modulacji ustawiony\"}");
            return;
        }
    }
    
    server.send(400, "application/json", "{\"status\":\"error\",\"message\":\"Nieprawidłowy typ modulacji\"}");
}

// Handler dla statusu
void handleStatus() {
    DynamicJsonDocument doc(256);
    doc["frequency"] = currentFrequency;
    doc["modulation"] = currentModulation == 0 ? "SPWM" : "SVPWM";
    doc["output_voltage"] = outputVoltage;
    doc["output_current"] = outputCurrent;
    doc["temperature"] = temperature;
    doc["is_running"] = systemRunning;
    
    String json;
    serializeJson(doc, json);
    server.send(200, "application/json", json);
}
